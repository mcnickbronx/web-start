# ===== Приложение: Django, API, задачи — везде, где оно запускается

# --- Режим и ключи

# Режим разработки: подробные ошибки на странице, вход в админку по HTTP, статику отдаёт
# само приложение. На сервере False
DEBUG=True

# Свои на каждом сервере, не короче 32 символов:
# python -c "import secrets; print(secrets.token_urlsafe(50))"
SECRET_KEY=django-insecure-change-me
# Подпись JWT в API. Свой ключ: смена SECRET_KEY не разлогинит API, и наоборот.
# Даже локально — не короче 32 символов, иначе PyJWT на каждый токен пишет предупреждение
JWT_SECRET_KEY=jwt-insecure-change-me-local-dev-only

# --- Домены

# Списки — через запятую. На сервере — настоящие домены
ALLOWED_HOSTS=*
CSRF_TRUSTED_ORIGINS=http://localhost:8000
# Домены веб-фронтов, которым разрешены запросы к API из браузера. Фронта нет — пусто
CORS_ALLOWED_ORIGINS=http://localhost:3000,http://localhost:5173

# --- База

# DEV=True — берётся dev-база (DEV_DB_*), DEV=False — DB_*. Читается только выбранная.
# Сервер с инфраструктурой: доступ кладёт сюда администратор (sudo project-db-env <проект> [DEV_]) —
# прод: DEV=False и DB_*, dev-сервер: DEV=True и DEV_DB_*. Подробнее — docs/deploy.md.
# Сменили пароль базы — та же команда и пересоздать контейнеры.
DEV=True

# Прод. На ноутбуке можно оставить пустым: так пароль прода не лежит у разработчика
DB_NAME=
DB_USER=
DB_PASSWORD=
DB_HOST=
DB_PORT=5432

# Dev: своя локальная база или dev-база сервера через SSH-туннель
# (тогда DEV_DB_HOST=127.0.0.1 и порт туннеля, например 15432)
DEV_DB_NAME=project
DEV_DB_USER=postgres
DEV_DB_PASSWORD=
DEV_DB_HOST=localhost
DEV_DB_PORT=5432

# Соединений в пуле на один процесс API: API_WORKERS × пул ≤ max_connections Postgres
API_DB_POOL_SIZE=10

# --- Очередь задач

# Пусто — задачи выполняются сразу в процессе и кэша нет (core/cache.py), Redis не нужен.
# На сервере с Redis из compose.yml (профиль full): redis://redis:6379/0
REDIS_URL=

# --- Почта

# Локально письма печатаются в консоль. На сервере — SMTP:
# EMAIL_BACKEND=django.core.mail.backends.smtp.EmailBackend
EMAIL_BACKEND=django.core.mail.backends.console.EmailBackend
EMAIL_HOST=smtp.example.com
EMAIL_PORT=587
EMAIL_HOST_USER=
EMAIL_HOST_PASSWORD=
EMAIL_USE_TLS=True
DEFAULT_FROM_EMAIL=noreply@example.com

# Адрес сайта (фронта), без пути: ссылки в письмах ведут сюда, и по нему форма узнаёт
# переходы внутри своего сайта
SITE_URL=http://localhost:3000
SITE_NAME=Проект

# ===== Docker compose — только на сервере; локально и в CI не читаются

# Образ, который выкладывает сервер (docs/deploy.md): dev-проект — ghcr.io/<владелец>/web-start-back:dev,
# сайт — ghcr.io/<владелец>/web-start-back:<полный хеш коммита main>
DEPLOY_IMAGE=

# Имя проекта — как его папка в /srv/projects, домен — тот, что указан проекту в инфраструктуре
COMPOSE_PROJECT_NAME=project
DOMAIN=back.example.com

# Профиль (docs/deploy.md): light — админка, API и задачи одним процессом (контентный сайт); full —
# админка отдельным процессом, задачи через Redis и воркер (кабинеты; тогда и REDIS_URL выше)
COMPOSE_PROFILES=light

# Процессов приложения — по числу ядер проекта (обычно половина сервера), сайту-визитке хватает
# одного. Процессов × размер пула Tortoise (API_DB_POOL_SIZE) ≤ max_connections Postgres.
# Только в full: ADMIN_WORKERS — процессы админки, TASK_WORKERS — воркера задач
API_WORKERS=1
ADMIN_WORKERS=1
TASK_WORKERS=1
